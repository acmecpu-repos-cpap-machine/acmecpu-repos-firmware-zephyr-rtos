/**********SLEEP IN***************/
void EnterSleep(void)
{
	Write(Command ,0x28);
	Write(Command ,0x10);
	Delay(120);
}

/**********SLEEP OUT***************/
void ExitSleep(void)
{
	Write(Command ,0x11);
	Delay(120);
	Write(Command ,0x29);
}
